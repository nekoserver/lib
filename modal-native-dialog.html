Popover API はポップオーバーの表示を宣言的に制御できる新しい Web API です。popover 属性と popovertarget 属性だけで動作します。

コードスニペット

基本 <dialog> 
<dialog id="myDialog">
  <div class="dialog-header">
    <h2>タイトル</h2>
    <button onclick="myDialog.close()">&times;</button>
  </div>
  <div class="dialog-body">内容</div>
  <div class="dialog-footer">
    <button onclick="myDialog.close()">閉じる</button>
  </div>
</dialog>

<button onclick="myDialog.showModal()">開く</button>

/* showModal() = モーダル (::backdrop あり)
   show()      = 非モーダル (backdrop なし)
   close()     = 閉じる */

/* CSS: ::backdrop でオーバーレイをスタイリング */
dialog::backdrop {
  background: rgba(0,0,0,.5);
}

/* dialog[open] でアニメーション */
dialog[open] {
  animation: pop .3s ease-out;
}
@keyframes pop {
  from { opacity: 0; transform: scale(0.9); }
  to   { opacity: 1; transform: scale(1); }
}
form method="dialog" 
<!-- form method="dialog" で submit 時に dialog が自動で閉じる -->
<dialog id="formDialog">
  <form method="dialog">
    <label>名前</label>
    <input type="text" required>
    <button type="submit" value="ok">OK</button>
  </form>
</dialog>

<script>
formDialog.addEventListener('close', () => {
  console.log(formDialog.returnValue); // "ok"
});
</script>
バックドロップクリックで閉じる 
<script>
// dialog 要素自体をクリック = backdrop 領域
dialog.addEventListener('click', (e) => {
  if (e.target === dialog) dialog.close();
});
</script>
Popover API 
<!-- 宣言的な Popover -->
<button popovertarget="myPopover">開く</button>
<div popover id="myPopover">
  ポップオーバーの内容
</div>

<!-- 属性 -->
<!-- popover          = auto (ライトディスミス有効) -->
<!-- popover="manual"  = 手動制御のみ -->
<!-- popovertarget     = 対象のポップオーバーID -->
<!-- popovertargetaction = toggle / show / hide -->

<!-- JavaScript API -->
<script>
element.showPopover();
element.hidePopover();
element.togglePopover();
</script>
サイドパネル / ボトムシートの CSS 
/* サイドパネル: 右端に固定 */
#sidePanel {
  max-width: 360px; width: 360px;
  height: 100vh; max-height: 100vh;
  margin: 0; margin-left: auto;
  border-radius: 16px 0 0 16px;
}
#sidePanel[open] {
  animation: slide-left .3s ease-out;
}
@keyframes slide-left {
  from { transform: translateX(100%); }
  to   { transform: translateX(0); }
}

/* ボトムシート: 画面下部に固定 */
#bottomSheet {
  max-width: 100%; width: 100%;
  margin: 0; margin-top: auto;
  border-radius: 20px 20px 0 0;
  max-height: 60vh;
}
#bottomSheet[open] {
  animation: slide-up .35s ease-out;
}
@keyframes slide-up {
  from { transform: translateY(100%); }
  to   { transform: translateY(0); }
}
