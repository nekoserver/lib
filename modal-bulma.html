ステップウィザード JavaScript 
let currentStep = 1;
const totalSteps = 3;

function wizardNav(dir) {
  document.getElementById('step' + currentStep).style.display = 'none';
  document.getElementById('step-nav-' + currentStep).classList.remove('is-active');
  currentStep += dir;
  document.getElementById('step' + currentStep).style.display = '';
  document.getElementById('step-nav-' + currentStep).classList.add('is-active');
  document.getElementById('wizard-prev').disabled = (currentStep === 1);
  const nextBtn = document.getElementById('wizard-next');
  if (currentStep === totalSteps) {
    nextBtn.textContent = '完了';
    nextBtn.onclick = () => toggleModal('modal8');
  } else {
    nextBtn.textContent = '次へ';
    nextBtn.onclick = () => wizardNav(1);
  }
}
