コードスニペット: :target モーダル 
<!-- HTML -->
<a href="#my-modal">開く</a>

<div id="my-modal" class="modal-target-overlay">
  <div class="modal-target-box">
    <a href="#" class="close">&times;</a>
    <h2>タイトル</h2>
    <p>内容</p>
  </div>
</div>

/* CSS */
.modal-target-overlay {
  position: fixed; inset: 0;
  background: rgba(0,0,0,.5);
  display: flex; align-items: center; justify-content: center;
  opacity: 0; visibility: hidden;
  transition: opacity .3s, visibility .3s;
}
.modal-target-overlay:target {
  opacity: 1; visibility: visible;
}
コードスニペット: Checkbox Hack モーダル 
<input type="checkbox" id="cb-modal" class="modal-cb-toggle">
<label for="cb-modal" class="btn">開く</label>

<div class="modal-cb-overlay">
  <div class="modal-cb-box">
    <label for="cb-modal" class="close-label">&times;</label>
    <h2>タイトル</h2>
    <p>内容</p>
  </div>
</div>

/* CSS */
.modal-cb-toggle { display: none; }
.modal-cb-overlay {
  position: fixed; inset: 0;
  background: rgba(0,0,0,.5);
  opacity: 0; visibility: hidden;
  transition: opacity .3s;
}
.modal-cb-toggle:checked ~ .modal-cb-overlay {
  opacity: 1; visibility: visible;
}
コードスニペット: ボトムシート 
<input type="checkbox" id="bs" class="bottom-sheet-toggle">
<label for="bs">開く</label>
<label for="bs" class="bottom-sheet-backdrop"></label>
<div class="bottom-sheet">
  <h2>タイトル</h2>
  <p>内容</p>
  <label for="bs" class="btn">閉じる</label>
</div>

/* CSS */
.bottom-sheet {
  position: fixed; bottom: -100%;
  left: 0; right: 0;
  background: #fff;
  border-radius: 16px 16px 0 0;
  padding: 2rem;
  transition: bottom .4s ease;
}
.bottom-sheet-toggle:checked ~ .bottom-sheet {
  bottom: 0;
}
