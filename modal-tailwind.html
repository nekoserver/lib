基本モーダル (Alpine.js + Tailwind) 
<div x-data="{ open: false }">
  <button @click="open = true"
    class="px-5 py-2 bg-indigo-600 text-white rounded-lg">開く</button>

  <template x-if="open">
    <div class="fixed inset-0 z-50 flex items-center justify-center
                bg-black/50" @click.self="open = false">
      <div class="bg-white rounded-xl p-6 w-[90%] max-w-md shadow-xl">
        <h2 class="text-xl font-bold">タイトル</h2>
        <p class="text-gray-600 mt-2">内容</p>
        <button @click="open = false"
          class="mt-4 px-4 py-2 bg-indigo-600 text-white rounded-lg">
          閉じる
        </button>
      </div>
    </div>
  </template>
</div>
確認ダイアログ 
<div x-data="{ open: false }">
  <button @click="open = true"
    class="px-5 py-2 bg-red-600 text-white rounded-lg">削除</button>

  <template x-if="open">
    <div class="fixed inset-0 z-50 flex items-center justify-center bg-black/50">
      <div class="bg-white rounded-xl p-6 max-w-sm text-center shadow-xl">
        <!-- アイコン -->
        <div class="w-14 h-14 bg-red-100 rounded-full
                    flex items-center justify-center mx-auto mb-3">
          <svg class="w-8 h-8 text-red-600" ...>...</svg>
        </div>
        <h2 class="text-lg font-bold">本当に削除しますか？</h2>
        <p class="text-gray-500 text-sm">この操作は取り消せません。</p>
        <div class="flex gap-3 justify-center mt-4">
          <button @click="open=false" class="...">キャンセル</button>
          <button @click="open=false" class="...">削除する</button>
        </div>
      </div>
    </div>
  </template>
</div>
カスタムアニメーション (tailwind.config) 
tailwind.config = {
  theme: { extend: {
    animation: {
      'fade-in':   'fadeIn .3s ease-out',
      'slide-up':  'slideUp .3s ease-out',
      'zoom-in':   'zoomIn .25s ease-out',
      'bounce-in': 'bounceIn .5s ease-out',
    },
    keyframes: {
      fadeIn:   { from: { opacity: 0 }, to: { opacity: 1 } },
      slideUp:  { from: { transform: 'translateY(40px)', opacity: 0 },
                  to:   { transform: 'translateY(0)',     opacity: 1 } },
      zoomIn:   { from: { transform: 'scale(0.8)', opacity: 0 },
                  to:   { transform: 'scale(1)',    opacity: 1 } },
      bounceIn: {
        '0%':   { transform: 'scale(0.3)', opacity: 0 },
        '50%':  { transform: 'scale(1.05)', opacity: 1 },
        '70%':  { transform: 'scale(0.95)' },
        '100%': { transform: 'scale(1)' }
      },
    }
  }}
}
