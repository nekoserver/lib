コードスニペット

基本モーダル 
<!-- トリガー -->
<button class="btn btn-primary"
        data-bs-toggle="modal"
        data-bs-target="#myModal">開く</button>

<!-- モーダル -->
<div class="modal fade" id="myModal" tabindex="-1">
  <div class="modal-dialog">
    <div class="modal-content">
      <div class="modal-header">
        <h5 class="modal-title">タイトル</h5>
        <button class="btn-close" data-bs-dismiss="modal"></button>
      </div>
      <div class="modal-body">内容</div>
      <div class="modal-footer">
        <button class="btn btn-secondary" data-bs-dismiss="modal">閉じる</button>
        <button class="btn btn-primary">保存</button>
      </div>
    </div>
  </div>
</div>
静的バックドロップ 
<div class="modal fade" id="staticModal"
     data-bs-backdrop="static"
     data-bs-keyboard="false" tabindex="-1">
  <div class="modal-dialog">
    <div class="modal-content">
      <!-- 閉じるボタンなし = 明示的操作を強制 -->
      <div class="modal-header">
        <h5>利用規約</h5>
      </div>
      <div class="modal-body">...</div>
      <div class="modal-footer">
        <button class="btn btn-primary"
                data-bs-dismiss="modal">同意する</button>
      </div>
    </div>
  </div>
</div>
Offcanvas 
<button class="btn btn-info"
        data-bs-toggle="offcanvas"
        data-bs-target="#myOffcanvas">開く</button>

<div class="offcanvas offcanvas-end" id="myOffcanvas">
  <div class="offcanvas-header">
    <h5>タイトル</h5>
    <button class="btn-close" data-bs-dismiss="offcanvas"></button>
  </div>
  <div class="offcanvas-body">内容</div>
</div>

<!-- 方向: offcanvas-start / offcanvas-end / offcanvas-top / offcanvas-bottom -->
Toast 通知 
<div class="toast-container position-fixed bottom-0 end-0 p-3">
  <div class="toast" id="myToast">
    <div class="toast-header">
      <strong class="me-auto">通知</strong>
      <button class="btn-close" data-bs-dismiss="toast"></button>
    </div>
    <div class="toast-body">メッセージ</div>
  </div>
</div>

<script>
const toast = new bootstrap.Toast(document.getElementById('myToast'));
toast.show();
</script>
